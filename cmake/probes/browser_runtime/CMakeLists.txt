cmake_minimum_required(VERSION 3.24)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/.cmake/api/v1/query")
file(WRITE "${CMAKE_BINARY_DIR}/.cmake/api/v1/query/codemodel-v2" "")
get_filename_component(engine "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(CMAKE_TOOLCHAIN_FILE MATCHES "Emscripten\\.cmake$" OR CMAKE_SYSTEM_NAME STREQUAL "Emscripten")
  include("${engine}/cmake/NauEmscriptenSdk.cmake")
  if(NOT EXISTS "$ENV{EMSDK_NODE}")
    message(FATAL_ERROR "EMSDK_NODE is missing; dot-source the active SDK emsdk_env.ps1 before lifecycle validation.")
  endif()
  set(CMAKE_CROSSCOMPILING_EMULATOR "$ENV{EMSDK_NODE}" CACHE STRING "Use the activated SDK Node runtime" FORCE)
  set(EMSCRIPTEN_SYSTEM_PROCESSOR wasm32 CACHE STRING "Target architecture")
endif()
project(NauBrowserRuntime LANGUAGES C CXX)
list(APPEND CMAKE_MODULE_PATH "${engine}/cmake")
include(NauCommon)
include(NauModule)
include(platformSpec/NauPlatformSetup)
set(NAU_RUNTIME_PROFILE minimal)
set(NAU_MINIMAL_TESTS ON)
include(NauMinimalRuntime)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(NauEngineFolder NauBrowserRuntime)
include(NauGitVersion)
git_check_version()
file(STRINGS "${engine}/.nauversion" NAU_VERSION_STRING)
nau_set_version("${NAU_VERSION_STRING}")
enable_testing()
set(NAU_MINIMAL_GMOCK ON)
add_subdirectory("${engine}/engine/3rdparty_libs" dependencies)
add_subdirectory("${engine}/engine/core/kernel/src" kernel)

if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "The browser lifecycle fixture requires Emscripten")
endif()
add_subdirectory("${engine}/engine/core/modules/platform_app/src" platform)
add_subdirectory("${engine}/engine/core/app_framework/src" framework)
add_executable(NauBrowserRuntimeTests "${CMAKE_CURRENT_SOURCE_DIR}/fixture.cpp")
target_link_libraries(NauBrowserRuntimeTests PRIVATE NauFrameworkMinimal gtest)
nau_target_link_modules(NauBrowserRuntimeTests PlatformApp)
nau_add_compile_options(NauBrowserRuntimeTests)
target_link_options(NauBrowserRuntimeTests PRIVATE -sEXIT_RUNTIME=1 -sPTHREAD_POOL_SIZE=12 -sALLOW_BLOCKING_ON_MAIN_THREAD=0)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(scenario success immediate deferred stopstartup pending shutdownerror configfail setupfail hostloss stall batchfailure)
  add_test(NAME BrowserRuntime-${scenario} COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/run.py" --build "${CMAKE_BINARY_DIR}" --scenario ${scenario})
  set_tests_properties(BrowserRuntime-${scenario} PROPERTIES TIMEOUT 100 RUN_SERIAL TRUE)
endforeach()
foreach(scenario abort fixture-failure)
  add_test(NAME BrowserRuntime-${scenario} COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/verify_failure.py" ${scenario} "${CMAKE_BINARY_DIR}")
  set_tests_properties(BrowserRuntime-${scenario} PROPERTIES TIMEOUT 100 RUN_SERIAL TRUE)
endforeach()
