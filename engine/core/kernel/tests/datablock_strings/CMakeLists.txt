cmake_minimum_required(VERSION 3.24)
project(NauDataBlockStringTests LANGUAGES C CXX)
set(BUILD_SHARED_LIBS ON)
if(NOT NAU_ENGINE_SDK_DIR)
  message(FATAL_ERROR "Set NAU_ENGINE_SDK_DIR to a rebuilt Release DLL SDK")
endif()
get_filename_component(engine "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${NAU_ENGINE_SDK_DIR}/cmake")
include(NauCommon)
include(platformSpec/NauPlatformSetup)
set(CMAKE_CXX_STANDARD 20)
foreach(dependency fmt utf8cpp vectormath jsoncpp wyhash brotli lzma-9.20 zlib-ng zstd-1.4.5 md5 fast_float)
  include("${NAU_ENGINE_SDK_DIR}/cmake/3rdPartyLibs/${dependency}/${dependency}-config.cmake")
endforeach()
include("${NAU_ENGINE_SDK_DIR}/cmake/core/NauKernel/NauKernel-config.cmake")
set(gtest_force_shared_crt ON CACHE BOOL "Match the DLL SDK runtime" FORCE)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory("${engine}/engine/3rdparty_libs/googletest" googletest)
set(BUILD_SHARED_LIBS ON)
add_executable(NauDataBlockStringTests ../test_runtime/test_data_block.cpp)
target_precompile_headers(NauDataBlockStringTests PRIVATE ../test_runtime/pch.h)
target_link_libraries(NauDataBlockStringTests PRIVATE NauKernel gtest_main gmock)
nau_add_compile_options(NauDataBlockStringTests)
set_target_properties(NauDataBlockStringTests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
enable_testing()
add_test(NAME DataBlockStrings COMMAND NauDataBlockStringTests)
