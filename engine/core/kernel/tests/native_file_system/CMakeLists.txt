cmake_minimum_required(VERSION 3.24)
project(NauNativeFileSystemTests LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This regression exercises the Windows native filesystem")
endif()
get_filename_component(engine "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(NAU_ENGINE_SOURCE_DIR "${engine}")
list(APPEND CMAKE_MODULE_PATH "${engine}/cmake")
include(NauCommon)
include(NauModule)
include(platformSpec/NauPlatformSetup)
set(NAU_RUNTIME_PROFILE minimal)
include("${engine}/cmake/NauStandaloneMinimal.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(NauEngineFolder NauNativeFileSystemTests)
include(NauGitVersion)
git_check_version()
file(STRINGS "${engine}/.nauversion" NAU_VERSION_STRING)
nau_set_version("${NAU_VERSION_STRING}")
set(NAU_MINIMAL_TESTS ON)
add_subdirectory("${engine}/engine/3rdparty_libs" dependencies)
add_subdirectory("${engine}/engine/core/kernel/src" kernel)
target_sources(NauKernel PRIVATE "${engine}/engine/core/kernel/src/platform/windows/native_file_system/win_native_file_system.cpp")
add_executable(NauNativeFileSystemTests "${CMAKE_CURRENT_SOURCE_DIR}/../test_kernel/test_io/test_native_file_system.cpp")
target_link_libraries(NauNativeFileSystemTests PRIVATE NauKernel gtest_main)
nau_add_compile_options(NauNativeFileSystemTests)
enable_testing()
add_test(NAME NativeFileSystemUnicode COMMAND NauNativeFileSystemTests)
