cmake_minimum_required(VERSION 3.24)
if(CMAKE_TOOLCHAIN_FILE MATCHES "Emscripten.cmake$")
  get_filename_component(engine "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
  include("${engine}/cmake/NauEmscriptenSdk.cmake")
  set(CMAKE_CROSSCOMPILING_EMULATOR "$ENV{EMSDK_NODE}" CACHE STRING "SDK Node runtime" FORCE)
  set(EMSCRIPTEN_SYSTEM_PROCESSOR wasm32 CACHE STRING "Architecture")
endif()
project(NauStringMigrationTests LANGUAGES C CXX)
get_filename_component(engine "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(NAU_ENGINE_SOURCE_DIR "${engine}")
list(APPEND CMAKE_MODULE_PATH "${engine}/cmake")
include(NauCommon)
include(NauModule)
include(platformSpec/NauPlatformSetup)
set(NAU_RUNTIME_PROFILE minimal)
include("${engine}/cmake/NauStandaloneMinimal.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(NauEngineFolder NauStringMigrationTests)
include(NauGitVersion)
git_check_version()
file(STRINGS "${engine}/.nauversion" NAU_VERSION_STRING)
nau_set_version("${NAU_VERSION_STRING}")
set(NAU_MINIMAL_TESTS ON)
set(NAU_MINIMAL_GMOCK ON)
add_subdirectory("${engine}/engine/3rdparty_libs" dependencies)
add_subdirectory("${engine}/engine/core/kernel/src" kernel)
add_executable(NauStringMigrationTests test_values.cpp
  ../test_kernel/test_version.cpp
  ../test_wasm_core/test_std_migration.cpp
  ../test_runtime/test_string/test_construction.cpp
  ../test_runtime/test_string/test_operators.cpp
  ../test_runtime/test_string/test_string_view.cpp
  ../test_runtime/test_string/test_hash_string.cpp
  ../test_runtime/test_string/test_string_utils.cpp)
target_precompile_headers(NauStringMigrationTests PRIVATE ../test_runtime/pch.h)
target_link_libraries(NauStringMigrationTests PRIVATE gmock)
target_link_libraries(NauStringMigrationTests PRIVATE NauKernel gtest_main)
target_compile_definitions(NauStringMigrationTests PRIVATE NAU_MODULE_NAME=StringMigration)
nau_write_static_modules_initialization(moduleRegistration)
target_sources(NauStringMigrationTests PRIVATE "${moduleRegistration}")
nau_add_compile_options(NauStringMigrationTests)
enable_testing()
if(EMSCRIPTEN)
  target_link_options(NauStringMigrationTests PRIVATE -sEXIT_RUNTIME=1)
endif()
add_test(NAME StringMigrationValues COMMAND NauStringMigrationTests)
